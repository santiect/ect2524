\section{Motivação}
%------------------------------------------------------------

\begin{frame}
	\frametitle{Da solução construtiva à solução melhorada}
	\begin{itemize}
		\item Solução inicial: heurística construtiva;
		\item Como melhorá-la?
		\item \textbf{Busca local}: pequenas modificações, mantendo as que
			melhoram o custo;
		\item Resultado: um \textbf{ótimo local}, que pode coincidir ou não com
			o \textbf{ótimo global};
		\item Ótimo global = solução ótima.
	\end{itemize}
\end{frame}

		\item $x \in S$: uma solução;
		\item Função objetivo $f : S \to \mathbb{R}$;
		\item Mochila: $2^n$ soluções; TSP: $(n-1)!/2$ rotas.
	\end{itemize}

\begin{frame}
	\frametitle{Ótimo local e ótimo global}
	\begin{columns}[T]
	\begin{column}{0.48\textwidth}
	\begin{itemize}
		\item \textbf{Ótimo global}: $f(x^*) \geq f(x)$, $\forall x \in S$;
		\item Vizinhança $\mathcal{N}(\bar{x}) \subseteq S$: subconjunto do
			espaço de soluções, gerado por uma estratégia de modificação de
			$\bar{x}$;
		\item \textbf{Ótimo local} em relação a $\mathcal{N}$:
			$f(\bar{x}) \geq f(x)$, $\forall x \in \mathcal{N}(\bar{x})$;
		\item Todo ótimo global é ótimo local; o inverso não vale.
	\end{itemize}
	\end{column}

\section{Vizinhança e busca local}
%------------------------------------------------------------

\begin{frame}
	\frametitle{Vizinhança}
	\begin{block}{Definição}
		$\mathcal{N}(x)$: soluções obtidas de $x$ por uma pequena
		alteração (operador de movimento).
	\end{block}
	\begin{itemize}
		\item Vizinhança pequena: iteração barata, mais ótimos locais;
		\item Vizinhança grande: iteração cara, menos ótimos locais.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Efeito do tamanho da vizinhança}

	\end{tikzpicture}\\[1mm]
	{\scriptsize $x$ não é ótimo local}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Algoritmo de busca local}
	\begin{enumerate}
		\item $x \leftarrow$ solução inicial;
		\item Se existe $x' \in \mathcal{N}(x)$ melhor que $x$:
			$x \leftarrow x'$, repetir;
		\item Senão, parar: $x$ é ótimo local.
	\end{enumerate}
	\vspace{2mm}
	\begin{itemize}
		\item Sempre termina;
		\item Depende de $x$ inicial e de $\mathcal{N}$.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Escolha do vizinho}
	\begin{itemize}
		\item \textbf{Primeira aprimorante}: o primeiro $x'$ que melhora;
		\item \textbf{Melhor aprimorante}: o melhor de $\mathcal{N}(x)$;
		\item Podem parar em ótimos locais diferentes.
	\end{itemize}
\end{frame}

\section{Busca local na mochila}
%------------------------------------------------------------

\begin{frame}
	\frametitle{Problema da mochila}
	\begin{itemize}
		\item $n$ itens, cada um com peso $p_i$ e benefício $b_i$;
		\item Mochila com capacidade $C$;
		\item Decisão: $x_i = 1$ se o item $i$ é levado, $x_i = 0$ caso
			contrário.
	\end{itemize}
	\vspace{2mm}
	\begin{block}{Modelo}
		\vspace{-4mm}
		\begin{align*}
			\max \quad & \sum_{i=1}^{n} b_i x_i \\
			\text{s.a.} \quad & \sum_{i=1}^{n} p_i x_i \leq C \\
			& x_i \in \{0,1\}, \quad i = 1, \ldots, n
		\end{align*}
	\end{block}
\end{frame}

\begin{frame}
	\frametitle{Vizinhança: inversão de bit}
	\begin{center}
	\begin{tikzpicture}
		\node[anchor=east,font=\small] at (0.2,0.5) {solução $x$};
		\vetor{0.5}{1,0,1,1,1,0,0}
		\node[anchor=east,font=\small] at (0.2,-0.5) {inserir item 6};
		\vetor{-0.5}{1,0,1,1,1,1,0}
		\destaque{-0.5}{6}
	\end{tikzpicture}
	\end{center}
	\begin{itemize}
		\item Solução: $x \in \{0,1\}^n$, com $\sum_i p_i x_i \leq C$;
		\item Muda um único $x_i$ de $0$ para $1$: inserir um item;
		\item $|\mathcal{N}| = n - k$, com $k$ = número de itens na mochila;
		\item Só se o item couber em $C$;
		\item Só vizinhos viáveis.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Vizinhança: troca}
	\begin{center}
	\begin{tikzpicture}
		\node[anchor=east,font=\small] at (0.2,0.5) {solução $x$};
		\vetor{0.5}{1,0,1,1,1,0,0}
		\node[anchor=east,font=\small] at (0.2,-0.5) {trocar 4 por 6};
		\vetor{-0.5}{1,0,1,0,1,1,0}
		\destaque{-0.5}{4}
		\destaque{-0.5}{6}
	\end{tikzpicture}
	\end{center}
	\begin{itemize}
		\item Sai um item da mochila, entra um item de fora;
		\item $|\mathcal{N}| = k(n-k)$, com $k$ = número de itens na
			mochila;
		\item Remoção e inserção em um só passo: alcança soluções fora da
			inversão de bit;
		\item Só vizinhos viáveis.
	\end{itemize}
\end{frame}

\section{Busca local no TSP: 2-opt}
%------------------------------------------------------------

\begin{frame}
	\frametitle{2-opt: exemplo e efeito do movimento}
	\begin{itemize}
		\item Remove 2 arestas e reconecta, invertendo o trecho entre elas;
		\item $|\mathcal{N}| = n(n-3)/2$.
	\end{itemize}
	\vspace{1mm}
	\begin{center}
	\begin{tikzpicture}[scale=0.8,
		cid/.style={circle,draw,fill=blue!15,inner sep=1.5pt,font=\scriptsize}]
		\begin{scope}

\begin{frame}
	\frametitle{2-opt: inversão de um trecho de $k$ cidades}
	\begin{center}
	\begin{tikzpicture}
		\node[anchor=east,font=\small] at (0.2,1.5) {rota};
		\rotavec{1.5}{1,2,3,4,5,6,7,8}{0}{0}
		\node[anchor=east,font=\small] at (0.2,0.6) {$k=2$};
		\rotavec{0.6}{1,2,4,3,5,6,7,8}{3}{4}
		\node[anchor=east,font=\small] at (0.2,-0.3) {$k=3$};
		\rotavec{-0.3}{1,2,5,4,3,6,7,8}{3}{5}
		\node[anchor=east,font=\small] at (0.2,-1.2) {$k=4$};
		\rotavec{-1.2}{1,2,6,5,4,3,7,8}{3}{6}
	\end{tikzpicture}
	\end{center}
	\begin{itemize}
		\item Movimento 2-opt: inverter $k$ cidades consecutivas da rota;
		\item Demais posições não mudam.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Variação de custo de um movimento 2-opt}
	\begin{itemize}
		\item Arestas $(a,b)$ e $(c,d)$ saem; $(a,c)$ e $(b,d)$ entram;
		\item Variação de custo, em tempo constante:
			\[
				\Delta = d_{ac} + d_{bd} - d_{ab} - d_{cd}
			\]
		\item $\Delta < 0$: o movimento melhora a rota.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Lógica da busca local 2-opt}
	\begin{enumerate}
		\item Partir de uma rota inicial (construtiva);
		\item Para $k = 2, 3, \ldots, n-2$ (tamanho do trecho invertido):
			\begin{itemize}
				\item para cada subrota possível de tamanho $k$, calcular $\Delta$;
				\item se $\Delta < 0$: inverter o trecho, somar $\Delta$ ao
					custo e \textbf{reiniciar} em $k = 2$;
			\end{itemize}
		\item Se nenhum tamanho $k$ produziu melhora: parar.
	\end{enumerate}
	\vspace{2mm}
	\begin{itemize}
		\item A rota final é um ótimo local em relação ao 2-opt;
		\item Primeiro movimento que melhora é aplicado.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Construção + busca local}
	\begin{itemize}
		\item Construtiva $\to$ busca local 2-opt $\to$ ótimo local;
		\item GAP diminui, mas em geral não se anula.
	\end{itemize}
\end{frame}
